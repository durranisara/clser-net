import copy
import torch
from utils.metrics import compute_metrics
from training.losses import clser_loss

def run_epoch(model, loader, device, optimizer=None,
              lambda_attr=0.1, lambda_sep=0.1, margin=1.0):
    train = optimizer is not None
    model.train(train)
    total_loss, preds, gts = 0.0, [], []
    for F, y in loader:
        F, y = F.to(device), y.to(device)
        logits, z, a, protos = model(F)
        loss, _ = clser_loss(logits, z, protos, y,
                             lambda_attr, lambda_sep, margin)
        if train:
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
        total_loss += loss.item() * F.size(0)
        preds.append(logits.argmax(-1).detach().cpu())
        gts.append(y.detach().cpu())
    preds = torch.cat(preds); gts = torch.cat(gts)
    metrics = compute_metrics(gts, preds)
    metrics["loss"] = total_loss / len(loader.dataset)
    return metrics

def train_with_early_stopping(model, train_loader, val_loader, optimizer,
                              epochs, device, patience=10, **loss_kw):
    best, best_state, wait = -1.0, None, 0
    for ep in range(1, epochs + 1):
        tr = run_epoch(model, train_loader, device, optimizer, **loss_kw)
        va = run_epoch(model, val_loader, device, None, **loss_kw)
        print(f"epoch {ep:03d} | train acc {tr['accuracy']:.4f} "
              f"| val acc {va['accuracy']:.4f} | val loss {va['loss']:.4f}")
        if va["accuracy"] > best:
            best, best_state, wait = va["accuracy"], copy.deepcopy(model.state_dict()), 0
        else:
            wait += 1
            if wait >= patience:
                print("early stopping")
                break
    if best_state is not None:
        model.load_state_dict(best_state)
    return model
